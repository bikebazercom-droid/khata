  const receiptRef = useRef(null);

  const handleShareFlow = async () => {
    if (!receiptRef.current) return;
    try {
      const canvas = await html2canvas(receiptRef.current, { backgroundColor: '#ffffff', scale: 3, useCORS: true });
      const jpgUrl = canvas.toDataURL('image/jpeg', 0.95);

      // Force Download Image to Device Gallery instantly
      const link = document.createElement('a');
      link.href = jpgUrl;
      link.download = `Receipt_${customer.name}.jpg`;
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);

      // Silently open native sharing sheet if supported without throwing UI alerts
      const res = await fetch(jpgUrl);
      const blob = await res.blob();
