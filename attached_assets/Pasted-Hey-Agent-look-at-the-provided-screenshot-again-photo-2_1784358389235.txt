import jsPDF from 'jspdf';
import 'jspdf-autotable';

// Ensure you have loaded your base64 font inside VFS for Bangla rendering

const generateExactKhatabookPDF = (filteredData, businessName = "হাজারি গোল্ড") => {
  const doc = new jsPDF('p', 'mm', 'a4');
  const pageWidth = doc.internal.pageSize.width; 
  const currentDateStr = new Date().toLocaleDateString('bn-BD', { day: 'numeric', month: 'long', year: 'numeric' });

  // REGISTER BANGLA FONT ROUTINE
  // doc.addFileToVFS('BanglaFont.ttf', banglaFontBase64);
  // doc.addFont('BanglaFont.ttf', 'BanglaFont', 'normal');
  // doc.setFont('BanglaFont');

  // 1. TOP BLUE STRIP
  doc.setFillColor(0, 75, 160); 
  doc.rect(0, 0, pageWidth, 14, 'F');
  
  doc.setTextColor(255, 255, 255);
  doc.setFontSize(10);
  doc.text(businessName, 14, 9);
  doc.text("Khatabook", pageWidth - 30, 9);

  // 2. MAIN REPORT HEADINGS
  doc.setDrawColor(255, 255, 255);
  doc.setTextColor(30, 41, 59);
  doc.setFontSize(13);
  doc.text("গ্রাহক তালিকার রিপোর্ট - ফিল্টার করা", pageWidth / 2, 24, { align: "center" });
  
  doc.setFontSize(9);
  doc.setTextColor(100, 116, 139);
  doc.text(`(আজ পর্যন্ত - ${currentDateStr})`, pageWidth / 2, 30, { align: "center" });

  // CALCULATE SUMMARY BALANCES
  let totalGet = 0;
  let totalGive = 0;
  filteredData.forEach(item => {
    if (item.balance > 0) totalGet += item.balance;
    else if (item.balance < 0) totalGive += Math.abs(item.balance);
  });
  const netBalance = totalGet - totalGive;

  // 3. THREE-COLUMN TOP CARD PANEL
  const cardY = 36;
  const cardWidth = pageWidth - 28;
  const colWidth = cardWidth / 3;

  doc.setDrawColor(226, 232, 240);
  doc.setFillColor(255, 255, 255);
  doc.rect(14, cardY, cardWidth, 18, 'DF');

  // Dividers
  doc.line(14 + colWidth, cardY, 14 + colWidth, cardY + 18);
  doc.line(14 + (colWidth * 2), cardY, 14 + (colWidth * 2), cardY + 18);

  // Col 1 Content
  doc.setFontSize(8); doc.setTextColor(148, 163, 184);
  doc.text("আপনি পাবেন", 14 + (colWidth / 2), cardY + 5, { align: "center" });
  doc.setFontSize(10); doc.setTextColor(15, 23, 42);
  doc.text(`৳${totalGet.toFixed(2)}`, 14 + (colWidth / 2), cardY + 12, { align: "center" });

  // Col 2 Content
  doc.setFontSize(8); doc.setTextColor(148, 163, 184);
  doc.text("আপনি দেবেন", 14 + colWidth + (colWidth / 2), cardY + 5, { align: "center" });
  doc.setFontSize(10); doc.setTextColor(15, 23, 42);
  doc.text(`৳${totalGive.toFixed(2)}`, 14 + colWidth + (colWidth / 2), cardY + 12, { align: "center" });

  // Col 3 Content
  doc.setFontSize(8); doc.setTextColor(148, 163, 184);
  doc.text("মোট ব্যালেন্স", 14 + (colWidth * 2) + (colWidth / 2), cardY + 5, { align: "center" });
  doc.setFontSize(10); doc.setTextColor(220, 38, 38); // Highlighted Red text style
  doc.text(`৳${Math.abs(netBalance).toFixed(2)} ${netBalance >= 0 ? 'Dr' : 'Cr'}`, 14 + (colWidth * 2) + (colWidth / 2), cardY + 12, { align: "center" });

  // 4. DATA METRIC HEADER
  doc.setFontSize(9);
  doc.setTextColor(51, 65, 85);
  doc.text(`গ্রাহক সংখ্যা: ${filteredData.length} (প্রাপ্ত)`, 14, 62);

  // 5. DATA TABLE SCHEMATICS
  const columns = ["নাম", "ডিটেলস", "আপনি পাবেন", "আপনি দেবেন", "সংগ্রহের দিন"];
  
  const bodyRows = filteredData.map(item => [
    item.name || "N/A",
    item.mobile || "N/A",
    item.balance > 0 ? item.balance.toFixed(2) : "",
    item.balance < 0 ? Math.abs(item.balance).toFixed(2) : "",
    item.collectionDay || ""
  ]);

  // Append Footer Summary Row
  bodyRows.push([
    "সর্বমোট", 
    "", 
    totalGet.toFixed(2), 
    totalGive.toFixed(2), 
    ""
  ]);

  doc.autoTable({
    startY: 66,
    head: [columns],
    body: bodyRows,
    theme: 'grid',
    styles: { fontSize: 8.5, cellPadding: 3, textColor: [51, 65, 85] },
    headStyles: {
      fillColor: [248, 250, 252], 
      textColor: [15, 23, 42],
      lineWidth: 0.1,
      lineColor: [226, 232, 240]
    },
    columnStyles: {
      0: { cellWidth: 42 },
      1: { cellWidth: 42 },
      2: { cellWidth: 34 }, // Will handle custom coloring inside cell hook
      3: { cellWidth: 34 }, // Will handle custom coloring inside cell hook
      4: { cellWidth: 30 }
    },
    willDrawCell: (data) => {
      // Apply exact column color codes from Screenshot
      if (data.row.type === 'body' && data.row.index !== bodyRows.length - 1) {
        if (data.column.index === 2) {
          data.cell.styles.fillColor = [254, 242, 242]; // Light Red background block
        }
        if (data.column.index === 3) {
          data.cell.styles.fillColor = [240, 253, 250]; // Light Green background block
        }
      }
      // Formatting for the total row ("সর্বমোট")
      if (data.row.index === bodyRows.length - 1) {
        data.cell.styles.fillColor = [241, 245, 249]; // Gray bottom background
        data.cell.styles.fontStyle = 'bold';
      }
    }
  });

  // 6. TIMESTAMPS FOOTNOTE
  const finalY = doc.lastAutoTable.finalY + 8;
  doc.setFontSize(8);
  doc.setTextColor(148, 163, 184);
  const currentTimeStr = new Date().toLocaleTimeString('bn-BD', { hour: '2-digit', minute: '2-digit' });
  doc.text(`রিপোর্ট তৈরি হয়েছে : ${currentTimeStr} | ${currentDateStr}`, 14, finalY);

  doc.save(`Khatabook_Statement_Report_${new Date().toISOString().split('T')[0]}.pdf`);
};
