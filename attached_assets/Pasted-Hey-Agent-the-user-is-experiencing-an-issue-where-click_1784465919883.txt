router.delete('/api/businesses/:id', requireAuth, async (req, res) => {
  const businessId = req.params.id;

    const ownership = await db.select().from(userBusinessesTable)

    if (ownership.length === 0) {
      return res.status(403).json({ error: "You do not own this business khata" });
    }

    // 2. Perform Clean Database Wipe for this business inside a transaction
    await db.transaction(async (tx) => {
      // Delete all transactions linked to customers of this business
      // Delete all customers linked to this business
      // Delete user_businesses join table record
      // Finally, delete the business record from businesses table
      
      await tx.delete(transactionsTable).where(eq(transactionsTable.businessId, businessId));
