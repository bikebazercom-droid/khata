When they log back in using their same Gmail/phone via Clerk or the auth service, they must land into a completely brand-new, freshly-provisioned blank account with zero legacy data.

1. CREATE TOTAL ACCOUNT NUKE ROUTE ON BACKEND (`api-server`):
Inside your authentication or user routes, add a new global account delete endpoint. It must trace all content belonging to the authenticated `req.user.id` and delete everything inside a single atomic database transaction:

```typescript
router.delete('/api/user/account/nuke', requireAuth, async (req, res) => {
  const userId = req.user.id;

  try {
    await db.transaction(async (tx) => {
      // 1. Get all business IDs owned by this user via the userBusinessesTable
      const userBiz = await tx.select({ businessId: userBusinessesTable.businessId })
        .from(userBusinessesTable)
        .where(eq(userBusinessesTable.userId, userId));
      
      const bizIds = userBiz.map(b => b.businessId);

      if (bizIds.length > 0) {
        // 2. Fetch all party/customer IDs linked to any of these businesses
        const parties = await tx.select({ id: partiesTable.id })
          .from(partiesTable)
          .where(inArray(partiesTable.businessId, bizIds));
        
        const partyIds = parties.map(p => p.id);

        // 3. Delete all ledger entries / transactions if parties exist
        if (partyIds.length > 0) {
          await tx.delete(ledgerEntriesTable).where(inArray(ledgerEntriesTable.partyId, partyIds));
        }

        // 4. Delete all child references for these businesses
        await tx.delete(partiesTable).where(inArray(partiesTable.businessId, bizIds));
        await tx.delete(businessSettingsTable).where(inArray(businessSettingsTable.businessId, bizIds));
        await tx.delete(userBusinessesTable).where(inArray(userBusinessesTable.businessId, bizIds));
        
        // 5. Delete the primary business records themselves
        await tx.delete(businessesTable).where(inArray(businessesTable.id, bizIds));
      }

      // 6. Finally, purge the main user identity record from the users table completely
      await tx.delete(usersTable).where(eq(usersTable.id, userId));
    });

    return res.status(200).json({ success: true, message: "Account completely erased." });
  } catch (error) {
    console.error("TOTAL ACCOUNT NUKE FAILED:", error);
    return res.status(500).json({ error: "Failed to fully wipe out the account records." });
  }
});
