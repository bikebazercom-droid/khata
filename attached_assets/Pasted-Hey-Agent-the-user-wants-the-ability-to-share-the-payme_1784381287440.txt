const handleUniversalJpgShare = async () => {

      backgroundColor: '#ffffff',
      scale: 3, // Premium clarity for small text blocks
      useCORS: true,
      logging: false
    });

    // 2. Convert to raw JPG data stream
    const jpgDataUrl = canvas.toDataURL('image/jpeg', 0.98);
    
    // Fallback down-loader helper function
    const forceLocalDownload = (url) => {
      const downloadAnchor = document.createElement('a');

      downloadAnchor.download = `Payment_Request_${customer?.name || 'Customer'}.jpg`;
      document.body.appendChild(downloadAnchor);
      downloadAnchor.click();
      document.body.removeChild(downloadAnchor);

    // 3. Prepare Binary Blob and Javascript File Entity
    const base64Response = await fetch(jpgDataUrl);
    const rawBlob = await base64Response.blob();

    // 4. Trigger Native OS Multi-App Sharing Dialog Sheet
    if (navigator.canShare && navigator.canShare({ files: [systemImageFile] })) {
