import React, { useRef } from 'react';

    if (!receiptCardAreaRef.current) return;

      // Snapshot the parent wrapper, but html2canvas will ignore elements marked with 'data-html2canvas-ignore'
      const canvas = await html2canvas(receiptCardAreaRef.current, {
        backgroundColor: '#ffffff',
        scale: 3, 
        useCORS: true,
        logging: false,
        // Enforce exclusion of elements with the ignore flag
        ignoreElements: (element) => element.hasAttribute('data-html2canvas-ignore')

      downloadAnchor.href = jpgDataUrl;
      downloadAnchor.download = filename;
      document.body.appendChild(downloadAnchor);
      downloadAnchor.click();
      document.body.removeChild(downloadAnchor);

      // 2. CONVERT BASE64 DATA INTO RAW BUFFER FOR THE NATIVE SHEET
      const byteString = atob(jpgDataUrl.split(',')[1]);
      const ab = new ArrayBuffer(byteString.length);
      const ia = new Uint8Array(ab);
      for (let i = 0; i < byteString.length; i++) {
        ia[i] = byteString.charCodeAt(i);
      }
      const imageBlob = new Blob([ab], { type: 'image/jpeg' });
      const sharedPhotoFile = new File([imageBlob], filename, { 
        type: 'image/jpeg',
        lastModified: Date.now()

      // 3. LAUNCH THE NATIVE SMARTPHONE SHARE OVERLAY PANEL (Matches photo_2026-07-19_10-49-30_3.jpg)

            text: 'Banglakhata থেকে পাঠানো লেনদেনের ছবি।'

          return;
        } catch (shareErr) {
          console.log("Native system share wrapper dismissed.");
        }
      }

      // Fallback fallback link if native rules apply
