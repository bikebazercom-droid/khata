// --- BOTTOM POPUP MODAL COMPONENT WRITING TO JPG FOR WHATSAPP ---
const PaymentRequestPopup = ({ customer, onClose }) => {
  const receiptRef = useRef(null);
  const myContextName = "Shakil"; // Owner context name

  const handleWhatsAppJpgShare = async () => {
    if (!receiptRef.current) return;

    try {
      // 1. Capture the element view box cleanly into a virtual canvas space
      const canvas = await html2canvas(receiptRef.current, {
        backgroundColor: '#ffffff',
        scale: 2, // Enhances high-resolution rendering text values
        useCORS: true,
        logging: false
      });

      // 2. Export cleanly as an optimized image/jpeg (JPG) format data stream
      const jpgDataUrl = canvas.toDataURL('image/jpeg', 0.95);
      
      // 3. Transform the byte characters into standard Blob data
      const blobResponse = await fetch(jpgDataUrl);
      const rawBlob = await blobResponse.blob();
      
      // 4. Create an active system file payload structure
      const finalJpgFile = new File([rawBlob], `Request_From_${myContextName}.jpg', { type: 'image/jpeg' });

      // 5. Trigger the Mobile/Browser native share system directly directed onto WhatsApp
      if (navigator.canShare && navigator.canShare({ files: [finalJpgFile] })) {
        await navigator.share({
          files: [finalJpgFile],
          title: 'পেমেন্ট অনুরোধ রসিদ',
          text: `আপনাকে অর্থ প্রদানের জন্য ${myContextName}-এর কাছে অনুরোধ করুন।`
        });
      } else {
        // Fallback fallback: Trigger seamless desktop down-loader pipeline
        const fallbackAnchor = document.createElement('a');
        fallbackAnchor.href = jpgDataUrl;
        fallbackAnchor.download = `Request_From_${myContextName}.jpg`;
        document.body.appendChild(fallbackAnchor);
        fallbackAnchor.click();
        document.body.removeChild(fallbackAnchor);
        alert('সরাসরি শেয়ার সমর্থিত নয়। পেমেন্ট রসিদটির JPG ছবি ডাউনলোড হয়েছে, অনুগ্রহ করে এটি হোয়াটসঅ্যাপে শেয়ার করে দিন।');
      }
    } catch (err) {
      console.error('Error generating or processing the JPG file data string:', err);
    }
  };

  return (
    <div style={{ position: 'fixed', top: 0, left: 0, width: '100%', height: '100%', backgroundColor: 'rgba(0,0,0,0.4)', zIndex: 9999, display: 'flex', alignItems: 'flex-end' }}>
      <div style={{ width: '100%', backgroundColor: '#ffffff', borderTopLeftRadius: '24px', borderTopRightRadius: '24px', padding: '20px 20px 24px 20px', boxSizing: 'border-box', animation: 'slideUp 0.3s ease-out' }}>
        
        {/* Dynamic Drag Notch Indicator bar */}
        <div style={{ width: '40px', height: '4px', backgroundColor: '#CBD5E1', borderRadius: '2px', margin: '0 auto 20px auto' }}></div>

        {/* --- DYNAMIC CAPTURING TARGET ZONE FOR EXTRACTING CLEAN JPG --- */}
        <div ref={receiptRef} style={{ backgroundColor: '#ffffff', padding: '10px 4px' }}>
          <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '16px' }}>
            <h4 style={{ margin: 0, fontSize: '15px', color: '#334155', fontWeight: 'bold' }}>
              আপনাকে অর্থ প্রদানের জন্য <span style={{ color: '#1E293B', fontWeight: '800' }}>{myContextName}</span>-এর কাছে অনুরোধ করুন
            </h4>
            <button onClick={onClose} style={{ border: 'none', background: '#F1F5F9', color: '#64748B', width: '28px', height: '28px', borderRadius: '50%', cursor: 'pointer', fontSize: '14px', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>✕</button>
          </div>

          <div style={{ marginBottom: '24px' }}>
            <div style={{ fontSize: '38px', fontWeight: 'bold', color: '#DC2626', letterSpacing: '-0.5px' }}>
              ৳ {customer.amount}
            </div>
            <div style={{ fontSize: '13px', color: '#94A3B8', marginTop: '2px' }}>দেবেন</div>
          </div>

          <div style={{ backgroundColor: '#F0F7FF', borderRadius: '12px', border: '1px solid #E0EFFF', padding: '14px 16px', display: 'flex', justifyContent: 'space-between', alignItems: 'center', cursor: 'pointer' }}>
