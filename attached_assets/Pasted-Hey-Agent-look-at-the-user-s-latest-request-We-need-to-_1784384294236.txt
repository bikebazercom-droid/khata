const executeNativeJpgShare = async (elementRef, filenameLabel) => {
  if (!elementRef.current) return;

    const canvas = await html2canvas(elementRef.current, {

      scale: 3, // Premium high-res scale for crisp fonts
      useCORS: true,
      logging: false
    });

    // 2. Convert canvas context directly to an absolute JPEG stream
    const jpgDataUrl = canvas.toDataURL('image/jpeg', 0.95);

    // 3. Fallback: Force a silent download to the phone storage as a safety net
    const phantomAnchor = document.createElement('a');
    phantomAnchor.href = jpgDataUrl;
    phantomAnchor.download = `${filenameLabel}_${new Date().getTime()}.jpg`;

    phantomAnchor.click();

    // 4. MODERN BASE64 TO BINARY FILE OBJECT STREAMING
    const base64Fetch = await fetch(jpgDataUrl);

    // 5. TRIGGER PHONE'S NATIVE OVERLAY SHARE SHEET INTERFACE
    if (navigator.canShare && navigator.canShare({ files: [sharedFile] })) {
